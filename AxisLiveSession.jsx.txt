// ─── PPG ENGINE (inline — sem dependências externas) ──────────────────────────
class PPGEngine {
  private buf: number[] = [];
  private tsBuf: number[] = [];
  private readonly WIN = 12000;
  private readonly MIN_GAP = 350;
  private readonly MAX_GAP = 2000;

  push(g: number, ts: number) {
    this.buf.push(g);
    this.tsBuf.push(ts);
    const cut = ts - this.WIN;
    while (this.tsBuf[0] < cut) { this.buf.shift(); this.tsBuf.shift(); }
  }

  private smooth(d: number[], w = 5) {
    return d.map((_, i) => {
      const s = Math.max(0, i - Math.floor(w/2));
      const e = Math.min(d.length, i + Math.ceil(w/2));
      const sl = d.slice(s, e);
      return sl.reduce((a, b) => a + b, 0) / sl.length;
    });
  }

  private peaks(data: number[], ts: number[]): number[] {
    const sm = this.smooth(data, 5);
    const mean = sm.reduce((a, b) => a + b, 0) / sm.length;
    const std = Math.sqrt(sm.reduce((a, b) => a + (b - mean) ** 2, 0) / sm.length);
    const thr = mean + std * 0.3;
    const out: number[] = [];
    let last = -Infinity;
    for (let i = 2; i < sm.length - 2; i++) {
      if (sm[i] > thr && sm[i] > sm[i-1] && sm[i] > sm[i-2] && sm[i] > sm[i+1] && sm[i] > sm[i+2] && ts[i] - last > this.MIN_GAP) {
        out.push(ts[i]); last = ts[i];
      }
    }
    return out;
  }

  get bpm(): number | null {
    if (this.buf.length < 30) return null;
    const pk = this.peaks(this.buf, this.tsBuf);
    if (pk.length < 2) return null;
    const rr = pk.slice(1).map((v, i) => v - pk[i]).filter(v => v >= this.MIN_GAP && v <= this.MAX_GAP);
    if (!rr.length) return null;
    return Math.round(60000 / (rr.reduce((a, b) => a + b, 0) / rr.length));
  }

  get rmssd(): number | null {
    if (this.buf.length < 60) return null;
    const pk = this.peaks(this.buf, this.tsBuf);
    if (pk.length < 3) return null;
    const rr = pk.slice(1).map((v, i) => v - pk[i]).filter(v => v >= this.MIN_GAP && v <= this.MAX_GAP);
    if (rr.length < 2) return null;
    const sq = rr.slice(1).map((v, i) => (v - rr[i]) ** 2);
    return Math.round(Math.sqrt(sq.reduce((a, b) => a + b, 0) / sq.length));
  }

  get coherence(): number | null {
    const r = this.rmssd;
    return r ? Math.max(10, Math.min(99, Math.round((r / 70) * 100))) : null;
  }

  get stress(): number | null {
    const r = this.rmssd, b = this.bpm;
    if (!r || !b) return null;
    return Math.max(0, Math.min(100, Math.round(100 - (r/80)*60 + ((b-50)/80)*40)));
  }

  signal(n = 150): { v: number; ts: number }[] {
    const sl = this.buf.slice(-n);
    const tsl = this.tsBuf.slice(-n);
    if (!sl.length) return [];
    const mn = Math.min(...sl), mx = Math.max(...sl), rng = mx - mn || 1;
    return sl.map((v, i) => ({ v: (v - mn) / rng, ts: tsl[i] }));
  }

  get count() { return this.buf.length; }
  reset() { this.buf = []; this.tsBuf = []; }
}

// ─── MINI WAVEFORM ────────────────────────────────────────────────────────────
function MiniWave({ signal, color, height = 48, width = 200 }: {
  signal: {v:number;ts:number}[];
  color: string;
  height?: number;
  width?: number;
}) {
  const ref = useRef<HTMLCanvasElement>(null);
  useEffect(() => {
    const c = ref.current; if (!c) return;
    const ctx = c.getContext("2d"); if (!ctx) return;
    const dpr = window.devicePixelRatio || 1;
    c.width = width * dpr; c.height = height * dpr;
    ctx.scale(dpr, dpr);
    ctx.clearRect(0, 0, width, height);
    ctx.fillStyle = A.bg1; ctx.fillRect(0, 0, width, height);
    if (signal.length < 2) {
      ctx.strokeStyle = `${color}33`; ctx.lineWidth = 1;
      ctx.beginPath(); ctx.moveTo(0, height/2); ctx.lineTo(width, height/2); ctx.stroke();
      return;
    }
    ctx.shadowColor = color; ctx.shadowBlur = 6;
    ctx.strokeStyle = color; ctx.lineWidth = 1.5; ctx.lineJoin = "round";
    ctx.beginPath();
    signal.forEach((p, i) => {
      const x = (i / (signal.length - 1)) * width;
      const y = height - p.v * height * 0.75 - height * 0.12;
      i === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y);
    });
    ctx.stroke();
  }, [signal, color, height, width]);
  return <canvas ref={ref} style={{ width, height, display:"block" }} />;
}

// ─── BPM GAUGE ────────────────────────────────────────────────────────────────
function BPMGauge({ bpm }: { bpm: number | null }) {
  const min = 40, max = 180;
  const val = bpm ? Math.min(max, Math.max(min, bpm)) : null;
  const pct = val ? ((val - min) / (max - min)) * 100 : 0;
  const angle = -140 + (pct / 100) * 280;
  const color = !val ? A.textF : val < 60 ? A.blue : val < 100 ? A.green : val < 130 ? A.amber : A.red;

  const toXY = (deg: number, r: number) => {
    const rad = (deg - 90) * Math.PI / 180;
    return { x: 70 + r * Math.cos(rad), y: 70 + r * Math.sin(rad) };
  };

  const arcPath = (r: number, startDeg: number, endDeg: number) => {
    const s = toXY(startDeg, r), e = toXY(endDeg, r);
    const large = endDeg - startDeg > 180 ? 1 : 0;
    return `M ${s.x} ${s.y} A ${r} ${r} 0 ${large} 1 ${e.x} ${e.y}`;
  };

  return (
    <div style={{ position:"relative", width:140, height:100 }}>
      <svg width={140} height={100} viewBox="0 0 140 100">
        {/* Background arc */}
        <path d={arcPath(54, -50, 230)} fill="none" stroke={`${color}18`} strokeWidth={8} strokeLinecap="round" />
        {/* Value arc */}
        {val && <path d={arcPath(54, -50, -50 + (pct/100)*280)} fill="none" stroke={color} strokeWidth={8} strokeLinecap="round"
          style={{ filter:`drop-shadow(0 0 4px ${color}88)`, transition:"all 0.4s ease" }} />}
        {/* Needle */}
        {val && (() => {
          const tip = toXY(angle - 90, 38);
          return <line x1={70} y1={70} x2={tip.x} y2={tip.y} stroke={color} strokeWidth={2} strokeLinecap="round"
            style={{ transformOrigin:"70px 70px", transition:"all 0.4s ease" }} />;
        })()}
        <circle cx={70} cy={70} r={4} fill={color} />
        {/* Labels */}
        <text x={70} y={82} textAnchor="middle" fontFamily={disp} fontSize={22} fill={color}
          style={{ transition:"all 0.3s" }}>
          {val ?? "—"}
        </text>
        <text x={70} y={94} textAnchor="middle" fontFamily={mono} fontSize={8} fill={`${color}66`}>BPM</text>
        <text x={18} y={90} fontFamily={mono} fontSize={7} fill={A.textF}>40</text>
        <text x={106} y={90} fontFamily={mono} fontSize={7} fill={A.textF}>180</text>
      </svg>
    </div>
  );
}

// ─── HRV RING ────────────────────────────────────────────────────────────────
function HRVRing({ value, label, color, max = 100 }: {
  value: number | null; label: string; color: string; max?: number;
}) {
  const pct = value ? Math.min(100, (value / max) * 100) : 0;
  const r = 28, circ = 2 * Math.PI * r;
  const dash = (pct / 100) * circ;
  return (
    <div style={{ display:"flex", flexDirection:"column", alignItems:"center", gap:4 }}>
      <svg width={70} height={70} viewBox="0 0 70 70">
        <circle cx={35} cy={35} r={r} fill="none" stroke={`${color}18`} strokeWidth={5} />
        <circle cx={35} cy={35} r={r} fill="none" stroke={color} strokeWidth={5}
          strokeDasharray={`${dash} ${circ}`} strokeLinecap="round"
          transform="rotate(-90 35 35)"
          style={{ transition:"stroke-dasharray 0.8s ease", filter:`drop-shadow(0 0 4px ${color}66)` }} />
        <text x={35} y={32} textAnchor="middle" fontFamily={disp} fontSize={14} fill={color}>
          {value ?? "—"}
        </text>
        <text x={35} y={44} textAnchor="middle" fontFamily={mono} fontSize={8} fill={`${color}66`}>
          {value ? "ms" : ""}
        </text>
      </svg>
      <span style={{ fontFamily:mono, fontSize:9, color:`${color}88`, letterSpacing:"0.1em" }}>{label}</span>
    </div>
  );
}

// ─── COHERENCE FIELD ──────────────────────────────────────────────────────────
function CoherenceField({ coherence, bpm }: { coherence: number | null; bpm: number | null }) {
  const [tick, setTick] = useState(0);
  useEffect(() => { const t = setInterval(() => setTick(p => p + 1), 50); return () => clearInterval(t); }, []);

  const coh = coherence || 0;
  const rings = [20, 32, 44, 56];

  return (
    <div style={{ display:"flex", alignItems:"center", justifyContent:"center" }}>
      <svg width={140} height={140} viewBox="0 0 140 140">
        {rings.map((r, i) => {
          const opacity = (coh / 100) * (0.3 - i * 0.06);
          const dashOffset = (tick * (i % 2 === 0 ? 0.5 : -0.7)) % (2 * Math.PI * r);
          return (
            <circle key={r} cx={70} cy={70} r={r}
              fill="none"
              stroke={A.amber}
              strokeWidth={1}
              strokeOpacity={Math.max(0.04, opacity)}
              strokeDasharray={`${3 + i} ${6 + i * 2}`}
              strokeDashoffset={dashOffset}
              style={{ transition:"stroke-opacity 1s" }}
            />
          );
        })}
        {/* Center pulse */}
        <circle cx={70} cy={70} r={12}
          fill={`${A.amber}${Math.round((coh / 100) * 40 + 8).toString(16).padStart(2,"0")}`}
          stroke={A.amber} strokeWidth={1} strokeOpacity={0.5}
        />
        <text x={70} y={66} textAnchor="middle" fontFamily={disp} fontSize={14} fill={A.amber}>
          {coh || "—"}
        </text>
        <text x={70} y={78} textAnchor="middle" fontFamily={mono} fontSize={7} fill={`${A.amber}88`}>

        {bpm && (
          <text x={70} y={120} textAnchor="middle" fontFamily={mono} fontSize={8} fill={`${A.amber}55`}>
            {bpm} BPM
          </text>
        )}
      </svg>
    </div>
  );
}

// ─── ENGINE CONTROL ───────────────────────────────────────────────────────────
function EngineControl({ onCommand, sessionId }: {
  onCommand: (table: string, cmd: string, params: Record<string, unknown>) => void;
  sessionId: string;
}) {
  const [active, setActive]   = useState<string|null>(null);
  const [params, setParams]   = useState<Record<string, number>>({});
  const [elapsed, setElapsed] = useState(0);
  const [log, setLog]         = useState<string[]>([]);

  useEffect(() => {
    if (!active) { setElapsed(0); return; }
    const t = setInterval(() => setElapsed(p => p + 1), 1000);
    return () => clearInterval(t);
  }, [active]);

  function send(slug: string, cmd: string) {
    const tbl = TABLES.find(t => t.slug === slug);
    const p   = { ...params, session_id: sessionId, frequency_hz: tbl?.freq || 528 };
    const topic = `axis/${slug}/${cmd}`;
    onCommand(slug, cmd, p);
    setLog(prev => [`${new Date().toLocaleTimeString("pt-BR")} PUBLISH → ${topic}`, ...prev].slice(0, 8));
    if (cmd === "start") setActive(slug);
    if (cmd === "stop" && active === slug) setActive(null);
  }

  const fmt = (s: number) => `${String(Math.floor(s/60)).padStart(2,"0")}:${String(s%60).padStart(2,"0")}`;

  return (
    <div style={{ display:"flex", flexDirection:"column", gap:10 }}>

  const [hrv,       setHrv]       = useState<number|null>(null);
  const [coherence, setCoherence] = useState<number|null>(null);

  const [cohHistory,  setCohHistory]  = useState<number[]>([]);

  const [cmdLog, setCmdLog] = useState<string[]>([]);

    if (sessionState !== "active") return;
    const t = setInterval(() => setElapsed(Math.floor((performance.now() - startRef.current) / 1000)), 1000);
    return () => clearInterval(t);
  }, [sessionState]);

  // PPG frame processor
  const processFrame = useCallback(() => {
    const v = videoRef.current, c = canvasRef.current;
    if (!v || !c || v.readyState < 2) { rafRef.current = requestAnimationFrame(processFrame); return; }
    const ctx = c.getContext("2d", { willReadFrequently:true });
    if (!ctx) return;
    c.width = v.videoWidth; c.height = v.videoHeight;
    ctx.drawImage(v, 0, 0);
    const roiW = 80, roiH = 80;
    const roiX = Math.floor((v.videoWidth - roiW) / 2);
    const roiY = Math.floor((v.videoHeight - roiH) / 2);
    const img = ctx.getImageData(roiX, roiY, roiW, roiH).data;
    let sR = 0, sG = 0, n = 0;
    for (let i = 0; i < img.length; i += 4) { sR += img[i]; sG += img[i+1]; n++; }
    const avgG = sG / n, avgR = sR / n;
    const q: "good"|"poor"|"idle" = avgR > 100 && avgG > 40 && avgG < 200 ? "good" : avgR > 60 ? "poor" : "idle";
    setPpgQuality(q);

    if (engineRef.current.count % 15 === 0 && engineRef.current.count > 30) {
      const nb = engineRef.current.bpm;
      const nh = engineRef.current.rmssd;
      const nc = engineRef.current.coherence;
      const ns = engineRef.current.stress;
      if (nb && nb >= 40 && nb <= 180) {
        setBpm(prev => {

    }

    setPpgSignal(engineRef.current.signal(120));
    rafRef.current = requestAnimationFrame(processFrame);
  }, []);

  async function startCamera() {
    try {
      const stream = await navigator.mediaDevices.getUserMedia({
        video: { width:{ideal:640}, height:{ideal:480}, facingMode:"user", frameRate:{ideal:30} },
        audio: false,
      });
      streamRef.current = stream;
      if (videoRef.current) { videoRef.current.srcObject = stream; await videoRef.current.play(); }
      engineRef.current.reset();
      setCameraOn(true); setCameraErr("");
      rafRef.current = requestAnimationFrame(processFrame);
    } catch(e: unknown) {
      const msg = e instanceof Error ? e.message : "Erro";
      setCameraErr(msg);
    }

              <div style={{ background:A.bg2, border:`1px solid ${A.border}`, borderRadius:6, padding:"12px 14px" }}>
                <span style={{ fontSize:8, color:A.amberD, letterSpacing:"0.12em", display:"block", marginBottom:10 }}>TENDÊNCIA DA SESSÃO</span>
                <div style={{ display:"flex", flexDirection:"column", gap:8 }}>
                  {bpmHistory.length > 3 && (
                    <div>
                      <span style={{ fontSize:8, color:`${A.red}88`, marginBottom:4, display:"block" }}>BPM</span>
                      <MiniWave
                        signal={bpmHistory.map((v, i) => ({ v: (v - 40) / 140, ts: i }))}
                        color={A.red} height={28} width={280}
                      />
                    </div>
                  )}
                  {cohHistory.length > 3 && (
                    <div>
                      <span style={{ fontSize:8, color:`${A.green}88`, marginBottom:4, display:"block" }}>COERÊNCIA</span>
                      <MiniWave
                        signal={cohHistory.map((v, i) => ({ v: v / 100, ts: i }))}
                        color={A.green} height={28} width={280}
                      />
                    </div>
                  )}
                </div>
              </div>
            )}

            {/* Session log */}
            {cmdLog.length > 0 && (
              <div style={{ background:A.bg2, border:`1px solid ${A.border}`, borderRadius:6, padding:"10px 12px" }}>
                <span style={{ fontSize:8, color:A.amberD, letterSpacing:"0.12em", display:"block", marginBottom:6 }}>ENGINE LOG</span>
                {cmdLog.map((l, i) => (
                  <div key={i} style={{ fontSize:9, color:i===0?A.amberD:A.textF, lineHeight:1.7 }}>{l}</div>
                ))}
              </div>
            )}
          </div>
        </div>
      </div>
    </div>
  );
}
