// CONFIGURE: substitua pelos seus valores
const SUPABASE_URL  = (typeof window !== "undefined" && window.ENV_SUPABASE_URL)  || "https://SEU-PROJETO.supabase.co";
const SUPABASE_ANON = (typeof window !== "undefined" && window.ENV_SUPABASE_ANON) || "SUA-ANON-KEY";

async function sbFetch(path, opts = {}) {
  const token = localStorage.getItem("axis_token");
  const res = await fetch(`${SUPABASE_URL}/rest/v1${path}`, {
    ...opts,
    headers: {
      "apikey": SUPABASE_ANON,
      "Content-Type": "application/json",
      "Prefer": "return=representation",
      ...(token ? { "Authorization": `Bearer ${token}` } : {}),
      ...(opts.headers || {}),
    },
  });
  if (res.status === 204) return null;
  const data = await res.json();
  if (!res.ok) throw new Error(data.message || data.error_description || `HTTP ${res.status}`);
  return data;
}

async function sbAuth(email, password) {
  const res = await fetch(`${SUPABASE_URL}/auth/v1/token?grant_type=password`, {
    method: "POST",
    headers: { "apikey": SUPABASE_ANON, "Content-Type": "application/json" },
    body: JSON.stringify({ email, password }),
  });
  const data = await res.json();
  if (!res.ok) throw new Error(data.error_description || "Falha no login");
  return data;
}

function Alert({ type = "error", children }) {
  const styles = {
    error:   { bg:"rgba(239,68,68,0.1)",  border:"rgba(239,68,68,0.3)",  color:"#f87171" },
    success: { bg:"rgba(34,197,94,0.1)",  border:"rgba(34,197,94,0.3)",  color:"#4ade80" },
    info:    { bg:"rgba(99,102,241,0.1)", border:"rgba(99,102,241,0.3)", color:"#a5b4fc" },
  };
  const s = styles[type];
  return (
    <div style={{ padding:"10px 14px", borderRadius:8, background:s.bg, border:`1px solid ${s.border}`, fontSize:12, color:s.color, fontFamily:"monospace" }}>
      {children}
    </div>
  );
}

function RoleRouter({ user, onSignOut }) {
  const [panel, setPanel]   = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    // Pequeno delay para transição suave
    const t = setTimeout(() => {
      setPanel(user.role);
      setLoading(false);
    }, 300);
    return () => clearTimeout(t);
  }, [user.role]);

  if (loading) {
    const meta = ROLE_META[user.role] || ROLE_META.therapist;
    return (
      <div style={{ minHeight:"100vh", background:meta.bg, display:"flex", alignItems:"center", justifyContent:"center", flexDirection:"column", gap:16 }}>
        <Spinner color={meta.color} size={32} />
        <p style={{ fontFamily:"monospace", fontSize:11, color:`${meta.color}66`, letterSpacing:"0.12em" }}>
          CARREGANDO {meta.label.toUpperCase()}...
        </p>
      </div>
    );
  }

  // Injeta onSignOut e user em todos os painéis via window para simplicidade
  window.__axisUser     = user;
  window.__axisSignOut  = onSignOut;
  window.__axisGenPDF   = generateReportPDF;
  window.__axisSbFetch  = sbFetch;

  return <PanelSwitcher role={panel} user={user} onSignOut={onSignOut} />;
}

  switch (role) {
    case "admin":     return <AdminPanel     user={user} onSignOut={onSignOut} />;
    case "therapist": return <TherapistPanel user={user} onSignOut={onSignOut} />;
    case "client":    return <ClientPanel    user={user} onSignOut={onSignOut} />;
    case "student":   return <StudentPanel   user={user} onSignOut={onSignOut} />;
    default:          return <TherapistPanel user={user} onSignOut={onSignOut} />;
  }
}

                </div>
              </div>
            ))}
          </div>
        )}
      </div>
    </div>
  );
}

// ─── ROOT APP ─────────────────────────────────────────────────────────────────
export default function AxisCoreUnified() {
  const [user, setUser] = useState(() => {
    try { return JSON.parse(localStorage.getItem("axis_user")); } catch { return null; }
  });
  const [authChecked, setAuthChecked] = useState(false);

  useEffect(() => {
    // Verifica token existente
    const token = localStorage.getItem("axis_token");
    const savedUser = localStorage.getItem("axis_user");
    if (token && savedUser) {
      try {
        setUser(JSON.parse(savedUser));
      } catch { localStorage.clear(); }
    }
    setAuthChecked(true);
  }, []);

  function handleAuth(u) { setUser(u); }

  function handleSignOut() {
    localStorage.removeItem("axis_token");
    localStorage.removeItem("axis_user");
    setUser(null);
  }

  if (!authChecked) {
    return (
      <div style={{ minHeight:"100vh", background:"#080a0f", display:"flex", alignItems:"center", justifyContent:"center" }}>
        <Spinner color="#6366f1" size={28} />
      </div>
    );
  }

  if (!user) return <LoginScreen onAuth={handleAuth} />;

  return (
    <AuthContext.Provider value={{ user, signOut: handleSignOut }}>
      <RoleRouter user={user} onSignOut={handleSignOut} />
    </AuthContext.Provider>
  );
}
