    <span style={{
      display: "inline-flex", alignItems: "center", gap: 5,
      padding: "3px 10px", borderRadius: 99, fontSize: 11,
      fontFamily: "'DM Mono', monospace",

      border: `1px solid ${active ? "rgba(255,255,255,0.18)" : "rgba(255,255,255,0.05)"}`,
      color: active ? "rgba(255,255,255,0.75)" : "rgba(255,255,255,0.22)",

        {param.id}
      </span>
      <div style={{ textAlign: "right" }}>
        <span style={{ fontSize: 12, color: "rgba(255,255,255,0.8)" }}>
          {param.label}
        </span>
        {param.unit && (
          <span style={{ fontSize: 11, color: "rgba(255,255,255,0.3)", marginLeft: 4 }}>
            · {param.unit}
          </span>
        )}
        {param.options && (
          <div style={{ fontSize: 10, color: "rgba(255,255,255,0.3)", marginTop: 2, fontFamily: "'DM Mono', monospace" }}>
            [{param.options.join(" · ")}]
          </div>
        )}
      </div>
    </div>
  );
}

              <div style={{ display: "flex", flexDirection: "column", gap: 6 }}>
                {filtered.length === 0 ? (
                  <div style={{ padding: "24px 16px", textAlign: "center", color: "rgba(255,255,255,0.25)", fontSize: 12 }}>
                    Nenhum módulo encontrado
                  </div>
                ) : (
                  filtered.map(table => (
                    <TableCard
                      key={table.slug}
                      table={table}
                      isActive={selected?.slug === table.slug}
                      onClick={() => setSelected(table)}
                    />
                  ))
                )}
              </div>

              {/* Registry snippet */}
              <div style={{ marginTop: 16, padding: "14px", borderRadius: 10, background: "rgba(255,255,255,0.02)", border: "1px solid rgba(255,255,255,0.05)" }}>
                <div style={{ fontSize: 10, color: "rgba(255,255,255,0.25)", letterSpacing: "0.08em", marginBottom: 8 }}>
                  LOADER
                </div>
                <pre style={{ fontSize: 10, color: "rgba(255,255,255,0.45)", lineHeight: 1.7, whiteSpace: "pre-wrap" }}>{`await AxisTableLoader
  .loadDefaults();

registry.list()
  // ${TABLES.length} tables loaded`}</pre>
              </div>
            </div>

            {/* Right — detail panel */}
            {selected && (
              <DetailPanel
                table={selected}
                onClose={() => setSelected(null)}
              />
            )}

            {!selected && (
              <div style={{
                border: "1px dashed rgba(255,255,255,0.07)", borderRadius: 16,
                padding: "60px 40px", textAlign: "center",
                color: "rgba(255,255,255,0.2)", fontSize: 13,
              }}>
                Selecione um módulo para inspecionar
              </div>
            )}
          </div>

          {/* Footer */}
          <div style={{
            marginTop: 48, paddingTop: 20, borderTop: "1px solid rgba(255,255,255,0.05)",
            display: "flex", justifyContent: "space-between", alignItems: "center",
          }}>
            <span style={{ fontSize: 10, color: "rgba(255,255,255,0.2)", letterSpacing: "0.06em" }}>
              AXIS CORE™ · packages/engine/src · Extensível via JSON
            </span>
            <div style={{ display: "flex", gap: 16 }}>
              {["registry.ts","loader.ts","tables/"].map(f => (
                <span key={f} style={{
                  fontSize: 10, fontFamily: "'DM Mono', monospace",
                  color: "rgba(255,255,255,0.25)", letterSpacing: "0.04em",
                }}>
                  {f}
                </span>
              ))}
            </div>
          </div>
        </div>
      </div>

      <style>{`
        @keyframes pulse {
          0%, 100% { opacity: 1; }
          50% { opacity: 0.4; }
        }
      `}</style>
    </>
  );
}
