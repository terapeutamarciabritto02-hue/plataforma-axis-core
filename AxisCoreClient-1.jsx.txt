}

function Chip({ children, color=C.violet }) {
  return (
    <span style={{
      fontFamily:mono, fontSize:10, letterSpacing:"0.1em",

    </span>
  );
}

function ProgressRing({ value, size=60, stroke=4, color="#8b5cf6", label="" }) {
  const r = (size - stroke) / 2;
  const circ = 2 * Math.PI * r;
  const dash = (value / 100) * circ;
  return (
    <div style={{ display:"flex", flexDirection:"column", alignItems:"center", gap:4 }}>
      <svg width={size} height={size} style={{ transform:"rotate(-90deg)" }}>
        <circle cx={size/2} cy={size/2} r={r} fill="none" stroke="rgba(255,255,255,0.05)" strokeWidth={stroke} />
        <circle cx={size/2} cy={size/2} r={r} fill="none" stroke={color} strokeWidth={stroke}
          strokeDasharray={`${dash} ${circ}`} strokeLinecap="round"
          style={{ transition:"stroke-dasharray 1s ease" }} />
      </svg>
      <span style={{ fontFamily:mono, fontSize:9, color:C.textFaint, letterSpacing:"0.06em" }}>{label}</span>
    </div>
  );
}

function CoherenceArc({ value, color }) {
  const angle = (value / 100) * 180 - 90;
  const rad = (angle * Math.PI) / 180;
  const x = 60 + 50 * Math.cos(rad);
  const y = 60 + 50 * Math.sin(rad);
  const largeArc = value > 50 ? 1 : 0;
  const endAngle = ((value / 100) * 180 - 90) * Math.PI / 180;
  const endX = 60 + 50 * Math.cos(endAngle);
  const endY = 60 + 50 * Math.sin(endAngle);
  return (
    <svg width={120} height={70} viewBox="0 0 120 70">
      <path d="M 10 60 A 50 50 0 0 1 110 60" fill="none" stroke="rgba(255,255,255,0.06)" strokeWidth={6} strokeLinecap="round" />
      <path d={`M 10 60 A 50 50 0 ${largeArc} 1 ${endX} ${endY}`} fill="none" stroke={color} strokeWidth={6} strokeLinecap="round" style={{ transition:"all 1s ease" }} />
      <text x={60} y={52} textAnchor="middle" fontFamily={mono} fontSize={18} fill={color} fontWeight="700">{value}</text>
      <text x={60} y={64} textAnchor="middle" fontFamily={mono} fontSize={9} fill={C.textFaint}>%</text>
    </svg>
  );
}

// ─── CHAKRA MANDALA ───────────────────────────────────────────────────────────
function ChakraMandala({ levels = {}, size = 320, interactive = false }) {
  const [hovered, setHovered] = useState(null);
  const defaultLevels = { root:75, sacral:68, solar:80, heart:88, throat:72, third_eye:70, crown:76 };
  const lvls = { ...defaultLevels, ...levels };

  return (
    <div style={{ position:"relative", width:size, height:size }}>
      <svg width={size} height={size} viewBox="0 0 320 320">
        {/* Background rings */}
        {[40,70,100,130].map(r => (
          <circle key={r} cx={160} cy={160} r={r} fill="none" stroke="rgba(180,157,255,0.06)" strokeWidth={1} strokeDasharray="3 6" />
        ))}
        {/* Cross lines */}
        <line x1={160} y1={30} x2={160} y2={290} stroke="rgba(180,157,255,0.04)" />
        <line x1={30}  y1={160} x2={290} y2={160} stroke="rgba(180,157,255,0.04)" />

        {/* Body silhouette hint */}
        <ellipse cx={160} cy={160} rx={18} ry={120} fill="rgba(180,157,255,0.03)" stroke="rgba(180,157,255,0.06)" strokeWidth={1} />

        {/* Chakra nodes */}
        {CHAKRAS.map(ch => {
          const level = lvls[ch.id] ?? 70;
          const isHov = hovered === ch.id;
          const baseR = 10 + (level / 100) * 8;
          const glowR = baseR + 8;
          return (
            <g key={ch.id}
              onMouseEnter={() => interactive && setHovered(ch.id)}
              onMouseLeave={() => setHovered(null)}
              style={{ cursor: interactive ? "pointer" : "default" }}>
              {/* Outer glow */}
              <circle cx={ch.pos.cx} cy={ch.pos.cy} r={glowR}
                fill={`${ch.color}${isHov?"22":"10"}`}
                style={{ transition:"all 0.4s", filter:`blur(${isHov?4:2}px)` }} />
              {/* Main node */}
              <circle cx={ch.pos.cx} cy={ch.pos.cy} r={baseR}
                fill={`${ch.color}${isHov?"55":"33"}`}
                stroke={ch.color}
                strokeWidth={isHov ? 1.5 : 1}
                style={{ transition:"all 0.4s" }} />
              {/* Inner dot */}
              <circle cx={ch.pos.cx} cy={ch.pos.cy} r={3} fill={ch.color}
                style={{ animation:`pulse_${ch.id} 2s ease-in-out infinite` }} />
              {/* Level arc */}
              {(() => {
                const r2 = baseR + 14;
                const c2 = 2 * Math.PI * r2;
                const d2 = (level / 100) * c2;
                return (
                  <circle cx={ch.pos.cx} cy={ch.pos.cy} r={r2}
                    fill="none" stroke={`${ch.color}44`} strokeWidth={1.5}
                    strokeDasharray={`${d2} ${c2}`}
                    transform={`rotate(-90 ${ch.pos.cx} ${ch.pos.cy})`}
                    style={{ transition:"stroke-dasharray 1s ease" }} />
                );
              })()}
            </g>
          );
        })}

        {/* Hover tooltip */}
        {hovered && (() => {
          const ch = CHAKRAS.find(c => c.id === hovered);
          const level = lvls[hovered] ?? 70;
          const tx = ch.pos.cx > 160 ? ch.pos.cx - 90 : ch.pos.cx + 30;
          return (
            <g>
              <rect x={tx} y={ch.pos.cy - 22} width={88} height={44} rx={6}
                fill="rgba(17,15,36,0.95)" stroke={`${ch.color}44`} strokeWidth={1} />
              <text x={tx+44} y={ch.pos.cy-6} textAnchor="middle" fontFamily={mono} fontSize={11} fill={ch.color}>{ch.name}</text>
              <text x={tx+44} y={ch.pos.cy+8} textAnchor="middle" fontFamily={mono} fontSize={11} fill="rgba(230,220,255,0.8)">{level}% · {ch.hz}Hz</text>
            </g>
          );
        })()}
      </svg>
    </div>
  );
}

      case "chakras":  return <ChakraView sessions={SESSIONS_HISTORY} />;
      case "history":  return <History sessions={SESSIONS_HISTORY} />;
      case "schedule": return <Schedule client={CLIENT} />;
      case "reports":  return <ClientReports reports={REPORTS} sessions={SESSIONS_HISTORY} />;
      default:         return null;
    }
  }

  return (
    <div style={{ minHeight:"100vh", maxWidth:480, margin:"0 auto", background:C.bg, color:C.text, position:"relative" }}>
      <style>{`
        @import url('https://fonts.googleapis.com/css2?family=Cormorant+Garamond:ital,wght@0,300;0,400;0,600;1,300;1,400&family=DM+Mono:wght@300;400&family=Outfit:wght@300;400;600&display=swap');
        * { box-sizing:border-box; margin:0; padding:0; }
        body { background:${C.bg}; }
        ::-webkit-scrollbar { width:2px; }
        ::-webkit-scrollbar-thumb { background:${C.border2}; }
        button { outline:none; }
        textarea, input { outline:none; background:transparent; }
        @keyframes glow { 0%,100%{opacity:1}50%{opacity:0.4} }
      `}</style>

      {/* Ambient background */}
      <div style={{ position:"fixed", top:"-10%", left:"50%", transform:"translateX(-50%)", width:400, height:400, borderRadius:"50%", background:"radial-gradient(circle,rgba(139,92,246,0.06) 0%,transparent 65%)", pointerEvents:"none", zIndex:0 }} />
      <div style={{ position:"fixed", bottom:"10%", right:"-20%", width:300, height:300, borderRadius:"50%", background:"radial-gradient(circle,rgba(91,141,238,0.04) 0%,transparent 65%)", pointerEvents:"none", zIndex:0 }} />

      {/* Content */}
      <div style={{ position:"relative", zIndex:1 }}>
        {renderTab()}
      </div>

      <BottomNav active={tab} onChange={setTab} />
    </div>
  );
}
