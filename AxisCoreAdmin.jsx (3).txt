// ─── UTILS ────────────────────────────────────────────────────────────────────
const fmt = {
  num: n => n.toLocaleString("pt-BR"),
  pct: n => `${n.toFixed(1)}%`,
  uptime: () => {
    const s = Math.floor(Date.now()/1000) % 86400;
    return `${String(Math.floor(s/3600)).padStart(2,"0")}:${String(Math.floor((s%3600)/60)).padStart(2,"0")}:${String(s%60).padStart(2,"0")}`;
  },
};

// ─── UI ATOMS ─────────────────────────────────────────────────────────────────
function Cell({ children, style={}, onClick }) {
  return (
    <div onClick={onClick} style={{
      background: T.bg1, border:`1px solid ${T.border}`,
      padding:"14px 16px", ...style,
      cursor: onClick ? "pointer" : "default",
    }}>
      {children}
    </div>
  );
}

function Label({ children, style={} }) {
  return <span style={{ fontFamily:"monospace", fontSize:10, letterSpacing:"0.14em", color:T.textFaint, textTransform:"uppercase", ...style }}>{children}</span>;
}

function Num({ children, size=24, color=T.green, style={} }) {
  return <span style={{ fontFamily:"monospace", fontSize:size, fontWeight:700, color, lineHeight:1, ...style }}>{children}</span>;
}

function StatusDot({ status }) {
  const c = status==="active"||status==="ok" ? T.green : status==="inactive"||status==="err" ? T.red : T.amber;
  return <span style={{ display:"inline-block", width:6, height:6, borderRadius:"50%", background:c, boxShadow:`0 0 6px ${c}`, marginRight:6, flexShrink:0 }} />;
}

function Pill({ children, color=T.green }) {
  return (
    <span style={{
      fontFamily:"monospace", fontSize:10, letterSpacing:"0.1em", padding:"2px 8px",
      border:`1px solid ${color}44`, color, background:`${color}10`,
    }}>
      {children.toUpperCase()}
    </span>
  );
}

function THead({ cols }) {
  return (
    <div style={{ display:"grid", gridTemplateColumns:cols, padding:"6px 12px", background:T.bg3, borderBottom:`1px solid ${T.border2}` }}>
      {cols.split(" ").map((_,i,arr) => <div key={i} />)}
    </div>
  );
}

function Btn({ children, onClick, variant="primary", size="sm", disabled=false }) {
  const styles = {
    primary: { background:`${T.greenFaint}`, border:`1px solid ${T.green}44`, color:T.green },
    ghost:   { background:"transparent", border:`1px solid ${T.border2}`, color:T.textDim },
    danger:  { background:"rgba(255,68,68,0.08)", border:`1px solid ${T.red}44`, color:T.red },
    amber:   { background:"rgba(255,184,0,0.08)", border:`1px solid ${T.amber}44`, color:T.amber },
  };
  return (
    <button onClick={onClick} disabled={disabled} style={{
      fontFamily:"monospace", fontSize:11, letterSpacing:"0.08em",
      padding: size==="sm" ? "5px 12px" : "8px 18px",
      cursor: disabled ? "not-allowed" : "pointer",
      opacity: disabled ? 0.4 : 1,
      transition:"all 0.15s",
      display:"inline-flex", alignItems:"center", gap:6,
      ...styles[variant],
    }}>
      {children}
    </button>
  );
}

        {filtered.map(log => {

              <span style={{ fontSize:11, color: log.actor.includes("unknown")?T.red:log.actor.includes("system")?T.blue:T.textDim, alignSelf:"center" }}>{log.actor}</span>
              <span style={{ fontSize:11, color: ACTION_COLOR[cat]||T.textDim, alignSelf:"center" }}>{log.action}</span>
              <span style={{ fontSize:10, color:T.textFaint, alignSelf:"center" }}>{log.resource}</span>
              <span style={{ fontSize:10, color: log.ip==="—"?T.textFaint:T.textDim, alignSelf:"center" }}>{log.ip}</span>
              <div style={{ display:"flex", alignItems:"center", alignSelf:"center" }}>
                <StatusDot status={log.status==="ok"?"active":"inactive"} />
              </div>
            </div>
          );
        })}
      </div>
    </div>
  );
}
