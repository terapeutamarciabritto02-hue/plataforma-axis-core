// ─── TYPE BADGE ───────────────────────────────────────────────────────────────
function TypeBadge({ type }) {
  const map = {
    video: { icon:"▶", label:"Vídeo",   color:T.teal   },
    pdf:   { icon:"▦", label:"Apostila",color:T.sepia  },
    live:  { icon:"◎", label:"Ao Vivo", color:T.crimson },
    quiz:  { icon:"◈", label:"Avaliação",color:T.violet },
  };
  const m = map[type] || map.video;
  return (
    <span style={{ fontFamily:mono, fontSize:9, padding:"2px 7px", borderRadius:3,
      background:`${m.color}12`, border:`1px solid ${m.color}33`, color:m.color,
      letterSpacing:"0.06em", display:"inline-flex", alignItems:"center", gap:4 }}>
      {m.icon} {m.label.toUpperCase()}
    </span>
  );
}

  const [tab, setTab] = useState("library");
  const [selectedCourse, setSelectedCourse] = useState(null);

  function handleSelectCourse(course) {
    setSelectedCourse(course);
    setTab("my_courses");
  }

  function renderMain() {
    if (selectedCourse && tab === "my_courses") {
      return <CourseDetail course={selectedCourse} onBack={()=>setSelectedCourse(null)} />;
    }
    switch(tab) {
      case "library":    return <Library onSelectCourse={handleSelectCourse} />;
      case "my_courses": return <MyCourses onSelectCourse={c=>{setSelectedCourse(c);}} />;
      case "progress":   return <Progress />;
      case "certs":      return <Certificates />;
      default:           return null;
    }
  }

  return (
    <div style={{ minHeight:"100vh", display:"flex", fontFamily:body }}>
      <style>{`
        @import url('https://fonts.googleapis.com/css2?family=IM+Fell+English:ital@0;1&family=Cinzel+Decorative:wght@400;700&family=Libre+Baskerville:ital,wght@0,400;0,700;1,400&family=DM+Mono:wght@300;400&display=swap');
        * { box-sizing:border-box; margin:0; padding:0; }
        body { background:${T.ink}; }
        ::-webkit-scrollbar { width:5px; }
        ::-webkit-scrollbar-track { background:${T.warm}; }
        ::-webkit-scrollbar-thumb { background:${T.sepia}66; border-radius:0; }
        button { outline:none; }
        @keyframes shimmer { 0%,100%{opacity:0.6}50%{opacity:1} }
      `}</style>

      <Sidebar active={tab} onChange={t=>{setTab(t);if(t!=="my_courses")setSelectedCourse(null);}} student={STUDENT} />
      <div style={{ flex:1, display:"flex", flexDirection:"column", overflow:"hidden" }}>
        {renderMain()}
      </div>
    </div>
  );
}
